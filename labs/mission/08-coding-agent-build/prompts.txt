시리즈 8: 코딩 에이전트 구축
시리즈 개요
지금까지의 에이전트는 식당을 찾아 주고 예약을 잡아 주는 쪽이었습니다. 이 시리즈에서 만드는 에이전트는 코드를 씁니다 — 강남 다이닝 서비스의 예약 로직을 사람 대신 구현하는 코딩 에이전트입니다. 기성 코딩 에이전트를 부리는 법이 「코딩 에이전트 운용」이라면, 이 시리즈는 그 에이전트를 직접 만드는 쪽입니다.

전반부는 로컬에서 에이전트를 만들고 자기 교정 패턴을 완성합니다. Shell 실행과 파일 읽기·쓰기 도구를 직접 정의해 코딩 에이전트를 처음부터 만들고, Reviewer·Tester를 붙여 사람 없이 지적과 실행 결과로 코드를 교정하는 루프를 돌립니다. 후반부는 이 에이전트를 클라우드 서비스로 올립니다 — AgentCore Runtime에 배포해 Git 기반 팀 코딩 서비스로 만들고, Lambda MicroVMs로 코드 실행을 격리된 샌드박스에서 돌립니다.

python

# labs/dining-coder/tools.py
import subprocess
from pathlib import Path
from strands import tool

WORKSPACE = (Path(__file__).parent / "workspace").resolve()
WORKSPACE.mkdir(exist_ok=True)

DANGEROUS_COMMANDS = ["rm -rf /", "sudo", ":(){", "mkfs", "dd if="]

def _safe_path(path: str) -> Path | None:
    """workspace 안이면 해석된 경로, 밖이면 None."""
    target = (WORKSPACE / path).resolve()
    return target if target.is_relative_to(WORKSPACE) else None

@tool
def run_shell(command: str) -> str:
    """workspace를 작업 디렉터리로 셸 명령을 실행합니다."""
    if any(bad in command for bad in DANGEROUS_COMMANDS):
        return f"거부: 파괴적인 명령으로 판단했습니다 — {command}"
    try:
        result = subprocess.run(
            command, shell=True, cwd=WORKSPACE,
            capture_output=True, text=True, timeout=60
        )
    except subprocess.TimeoutExpired:
        return f"중단: 60초 실행 상한을 넘겨 종료했습니다 — {command}"
    return f"exit={result.returncode}\nstdout:\n{result.stdout}\nstderr:\n{result.stderr}"

@tool
def read_file(path: str) -> str:
    """workspace 안의 파일을 읽습니다."""
    target = _safe_path(path)
    if target is None:
        return f"거부: {path}는 workspace 밖입니다. workspace 안 상대 경로만 허용됩니다."
    if not target.exists():
        return f"실패: {path} 파일이 존재하지 않습니다."
    return target.read_text(encoding="utf-8")

@tool
def write_file(path: str, content: str) -> str:
    """workspace 안의 파일에 내용을 씁니다."""
    target = _safe_path(path)
    if target is None:
        return f"거부: {path}는 workspace 밖입니다. workspace 안 상대 경로만 허용됩니다."
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(content, encoding="utf-8")
    return f"작성 완료: {target.relative_to(WORKSPACE)} ({len(content)} bytes)"

if __name__ == "__main__":
    # 경계 테스트
    print(write_file("../escape.txt", "x"))       # 거부
    print(write_file("test.txt", "hello world"))  # 성공
    print(read_file("test.txt"))                  # 성공
    print(run_shell("rm -rf /"))                  # 거부
    print(run_shell("echo OK"))                   # 성공

터미널 — write_file("test.txt", "hello") 호출이 성공하고 workspace/test.txt 생성

# labs/dining-coder/coder.py
from strands import Agent
from strands.models import BedrockModel
from tools import run_shell, read_file, write_file

SYSTEM_PROMPT = """당신은 코드를 작성하는 에이전트입니다.

규칙:
1. workspace/ 안에서만 파일을 읽고·쓰고·실행합니다.
2. 파일을 작성한 뒤에는 반드시 run_shell로 실행해 결과를 확인합니다.
3. 실행이 실패하면 stderr에서 원인을 찾아 코드를 수정하고 다시 실행합니다.
4. 실행 성공이 확인될 때까지 완료를 선언하지 않습니다.
"""

model = BedrockModel(model_id="us.anthropic.claude-sonnet-4-6", region_name="us-west-2")
agent = Agent(
    model=model,
    system_prompt=SYSTEM_PROMPT,
    tools=[run_shell, read_file, write_file],
)

if __name__ == "__main__":
    result = agent(
        "예약 인원이 식당 최대 수용 인원을 넘는지 검증하는 함수를 "
        "workspace/reservation.py에 만들어 주세요. "
        "한우명가 30명, 트라토리아 벨라 20명, 스시 오마카세 하루 8명 기준입니다"
    )
    print(str(result))

# labs/dining-coder/app.py
import streamlit as st
from pathlib import Path
from coder import agent

WORKSPACE = Path(__file__).parent / "workspace"

st.set_page_config(page_title="코딩 에이전트 콘솔", page_icon="🤖")
st.title("🤖 코딩 에이전트 콘솔")
st.caption("코딩 요청을 입력하면 에이전트가 코드를 작성·실행합니다.")

if prompt := st.chat_input("코딩 요청을 입력하세요"):
    with st.chat_message("user"):
        st.write(prompt)

    with st.chat_message("assistant"):
        with st.spinner("에이전트가 작업 중..."):
            result = agent(prompt)
        st.write(str(result))

    # workspace 파일 목록·내용 표시
    st.subheader("📁 workspace/ 파일")
    files = sorted(WORKSPACE.rglob("*.py"))
    if not files:
        st.info("아직 생성된 파일이 없습니다.")
    for f in files:
        with st.expander(f"📄 {f.relative_to(WORKSPACE)}"):
            st.code(f.read_text(encoding="utf-8"), language="python")

도전 과제 방향
읽기 전용 모드 — READONLY=1 환경변수 설정 시 write_file과 run_shell이 거부 문자열을 반환하도록 분기
명령 화이트리스트 — ALLOWED_COMMANDS = ["python", "pytest", "ls"]로 제한하고 나머지 전부 거부
작업 로그 — Hook(AfterToolCallEvent)으로 도구명·인자·반환값을 workspace/.log 파일에 append
핵심 포인트
거부는 예외가 아니라 반환값 — 도구가 예외로 죽으면 에이전트는 이유를 모른 채 같은 시도를 반복합니다. 거부 이유를 문자열로 돌려주면 에이전트가 스스로 경로를 고쳐 재시도합니다.
경계는 해석된 절대 경로로 판정 — ".." 문자열 필터는 절대 경로와 심링크 우회를 놓칩니다. resolve() 후 is_relative_to()가 세 우회를 한 번에 막습니다.
`stderr`를 버리지 말 것 — 실행 실패의 원인은 대부분 stderr에 있고, 이 값이 곧 에이전트의 수정 근거입니다.
"쓴 뒤 실행하라"는 프롬프트에 명시 — 없으면 에이전트는 파일만 쓰고 검증 없이 완료를 선언합니다.
`timeout`은 선택이 아닙니다 — 대화형 명령이나 서버 실행 시도는 상한이 없으면 세션을 멈춥니다.

리뷰·테스트 자기 교정 루프
고급
Coder에 Reviewer와 Tester를 붙여, 지적과 실행 결과로 서로를 검증하며 코드를 완성하는 3에이전트 루프를 만듭니다.

미션 브리핑
선행 미션의 에이전트는 코드를 쓰고 실행까지 하지만, 자기 코드를 냉정하게 보지 못합니다. 경계값을 빼먹고도 "완료했습니다"라고 답하고, 실행이 통과하면 품질은 묻지 않습니다. 이번 미션에서는 검증을 다른 에이전트에게 맡깁니다 — Reviewer는 코드를 읽고 지적하고, Tester는 테스트를 써서 실행으로 증명합니다. 코디네이터는 두 판정을 모아 Coder에게 재작업을 지시하고, 둘 다 통과할 때까지 최대 3회 반복합니다. 사람이 개입하지 않는 자기 교정 루프를 만드는 것이 목표입니다.

Coder·Reviewer·Tester 세 에이전트를 @tool 함수로 감싸고, 파이썬 코디네이터가 이를 호출해 조율하게 하세요
Reviewer는 APPROVED 또는 NEEDS_CHANGES + 지적 사항으로, Tester는 마지막 줄 passed=N failed=M 형식의 테스트 실행 결과로 판정하세요
둘 중 하나라도 통과하지 못하면 Coder가 재작업 — 최대 3회로 제한하고, 초과 시 최선의 코드와 남은 지적을 함께 반환하세요

import logging
from strands import Agent, tool
from strands.models import BedrockModel
from tools import read_file, write_file, run_shell
from reviewer import reviewer
from tester import tester

logger = logging.getLogger(__name__)
logging.basicConfig(level=logging.INFO)

# --- Coder 에이전트 ---
coder = Agent(
    model=BedrockModel(model_id="us.anthropic.claude-sonnet-4-6", region_name="us-west-2"),
    system_prompt="당신은 파이썬 개발자입니다. workspace/ 디렉토리에 코드를 작성합니다.",
    tools=[read_file, write_file, run_shell],
)

@tool
def write_code(task: str) -> str:
    """Coder에게 코드 작성을 요청합니다."""
    return str(coder(task))

@tool
def review_code(path: str) -> str:
    """Reviewer에게 코드 품질 판정을 요청합니다."""
    return str(reviewer(f"{path} 파일의 코드를 검토해 주세요."))

@tool
def run_tests(path: str) -> str:
    """Tester에게 테스트 작성·실행을 요청합니다."""
    return str(tester(f"{path} 파일의 함수를 테스트해 주세요."))

def coordinate(task: str) -> dict:
    """코디네이터 — 1회 호출로 세 에이전트 순차 실행."""
    logger.info("=== 코디네이터 시작: %s ===", task)

    code = write_code(task)
    logger.info("round=1 Coder 완료")

    review = review_code("workspace/reservation.py")
    approved = str(review).strip().startswith("APPROVED")
    logger.info("round=1 approved=%s", approved)

    test_result = run_tests("workspace/reservation.py")
    test_output = str(test_result).strip()
    last_line = test_output.split("\n")[-1]
    passed = "failed=0" in last_line
    logger.info("round=1 passed=%s (%s)", passed, last_line)

    return {
        "approved": approved,
        "passed": passed,
        "review": str(review),
        "test": test_output,
    }

if __name__ == "__main__":
    result = coordinate(
        "예약 인원 검증 함수를 경계값과 잘못된 입력까지 처리하도록 만들어 주세요"
    )
    print(result)

python

MAX_ROUNDS = 3

def build_with_review(task: str) -> dict:
    """자기 교정 루프 — 최대 3회 재작업."""
    feedback = ""
    code = ""
    history = []

    for round_no in range(1, MAX_ROUNDS + 1):
        # Coder 호출 — 재작업 시 피드백 원문 전달
        prompt = task
        if feedback:
            prompt = f"{task}\n\n지난 회차 지적 사항 (원문 그대로):\n{feedback}"
        code = str(write_code(prompt))
        logger.info("round=%s Coder 완료", round_no)

        # Reviewer 판정
        review = str(review_code("workspace/reservation.py")).strip()
        approved = review.startswith("APPROVED")
        logger.info("round=%s approved=%s", round_no, approved)

        # Tester 판정
        test_output = str(run_tests("workspace/reservation.py")).strip()
        last_line = test_output.split("\n")[-1]
        passed = "failed=0" in last_line
        logger.info("round=%s passed=%s (%s)", round_no, passed, last_line)
        history.append({"round": round_no, "approved": approved,
                        "passed": passed, "review": review, "test": last_line})

        # 두 판정 모두 통과
        if approved and passed:
            return {"state": "APPROVED", "rounds": round_no,
                    "code": code, "history": history}

        # 미통과 — 원문으로 피드백 전달
        feedback = f"[리뷰 판정]\n{review}\n\n[테스트 결과]\n{test_output}"

    # 3회 초과 — 최선 결과 반환
    return {
        "state": "BEST_EFFORT",
        "history": history,
        "rounds": MAX_ROUNDS,
        "code": code,
        "remaining": feedback,
    }

if __name__ == "__main__":
    # 부족한 요구로 재작업 루프 확인
    result = build_with_review("인원이 수용 인원을 넘는지만 보면 됩니다")
    print(f"최종 상태: {result['state']} (라운드: {result['rounds']})")
확인

3회 초과 시 — {"state": "BEST_EFFORT", "remaining": ...} 반환, 무한 루프 없음
재작업 시 — Reviewer 지적 + Tester 실패 로그가 요약 없이 원문으로 Coder에게 전달됨

python

import streamlit as st
from orchestrator import build_with_review

st.set_page_config(page_title="코딩 리뷰 루프", page_icon="🔄")
st.title("🔄 리뷰·테스트 자기 교정 루프")

if prompt := st.chat_input("코딩 요청을 입력하세요"):
    with st.chat_message("user"):
        st.write(prompt)

    with st.chat_message("assistant"):
        with st.spinner("에이전트 루프 실행 중..."):
            result = build_with_review(prompt)

        # 최종 상태 표시
        if result["state"] == "APPROVED":
            st.success(f"✅ 통과 (라운드 {result['rounds']})")
        else:
            st.warning(f"⚠️ 최선 결과 (라운드 {result['rounds']} — 상한 초과)")
            st.text("남은 지적:")
            st.text(result.get("remaining", ""))

        # 라운드별 판정 — Reviewer·Tester 결과를 순서대로 표시
        for h in result["history"]:
            icon = "✅" if h["approved"] and h["passed"] else "🔁"
            st.write(f"{icon} 라운드 {h['round']} — Reviewer: "
                     f"{'APPROVED' if h['approved'] else 'NEEDS_CHANGES'} · Tester: {h['test']}")
            if not h["approved"]:
                with st.expander(f"라운드 {h['round']} 리뷰 지적"):
                    st.text(h["review"])

        # 최종 코드 표시
        st.subheader("최종 코드")
        try:
            with open("workspace/reservation.py") as f:
                st.code(f.read(), language="python")
        except FileNotFoundError:
            st.error("workspace/reservation.py를 찾을 수 없습니다.")

# --- CodingService/app/CodingService/main.py ---
from bedrock_agentcore import BedrockAgentCoreApp
from strands import Agent, tool
from strands.models import BedrockModel
import json
import subprocess
from pathlib import Path
from datetime import datetime, timezone

WORKSPACE = Path("/workspace")
PERSISTENT = Path("/mnt/persistent")

SYSTEM_PROMPT = """당신은 팀 코딩 에이전트입니다.
자연어 코딩 요청을 받아 코드를 작성하고 테스트합니다.
workspace 안에서만 작업하고, 완료 후 결과를 보고합니다."""

@tool
def run_shell(command: str) -> str:
    """workspace 안에서 셸 명령을 실행합니다."""
    blocked = ["rm -rf /", "shutdown", "reboot", "mkfs"]
    if any(b in command for b in blocked):
        return f"거부: 파괴적 명령 '{command}'"
    try:
        result = subprocess.run(
            command,
            shell=True,
            cwd=str(WORKSPACE),
            capture_output=True,
            text=True,
            timeout=120,
        )
        output = result.stdout + result.stderr
        return output if output.strip() else f"(exit {result.returncode})"
    except subprocess.TimeoutExpired:
        return "오류: 명령 실행 시간 초과 (120초)"

@tool
def read_file(path: str) -> str:
    """workspace 안의 파일을 읽습니다."""
    target = (WORKSPACE / path).resolve()
    if not target.is_relative_to(WORKSPACE):
        return f"거부: workspace 밖 경로 '{path}'"
    if not target.exists():
        return f"오류: 파일 없음 '{path}'"
    return target.read_text()

@tool
def write_file(path: str, content: str) -> str:
    """workspace 안에 파일을 씁니다."""
    target = (WORKSPACE / path).resolve()
    if not target.is_relative_to(WORKSPACE):
        return f"거부: workspace 밖 경로 '{path}'"
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(content)
    return f"저장 완료: {path}"

def append_work_log(request: str, result: str):
    """작업 로그를 영속 경로에 기록합니다."""
    log_path = PERSISTENT / "work-log.json"
    logs = json.loads(log_path.read_text()) if log_path.exists() else []
    logs.append({
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "request": request,
        "result": result[:500],
    })
    log_path.write_text(
        json.dumps(logs, ensure_ascii=False, indent=2)
    )

model = BedrockModel(
    model_id="us.anthropic.claude-sonnet-4-6",
    region_name="us-west-2",
)
agent = Agent(
    model=model,
    system_prompt=SYSTEM_PROMPT,
    tools=[run_shell, read_file, write_file],
)

app = BedrockAgentCoreApp()

@app.entrypoint
def my_agent(payload):
    prompt = payload.get("prompt", "")
    result = str(agent(prompt))
    append_work_log(request=prompt, result=result)
    return {"result": result}

# --- test_invoke.py ---
import boto3
import json
import sys
import uuid

agentcore_client = boto3.client(
    "bedrock-agentcore",
    region_name="us-west-2",
)

RUNTIME_ARN = "<YOUR_RUNTIME_ARN>"
# runtimeSessionId는 호출자가 만들어 전달하는 입력값입니다 (33자 이상)
SESSION_ID = f"coding-service-{uuid.uuid4().hex}"

def run_command(command: str, timeout: int = 300) -> int:
    """세션 안에서 셸 명령을 실행하고 종료 코드를 반환합니다."""
    response = agentcore_client.invoke_agent_runtime_command(
        agentRuntimeArn=RUNTIME_ARN,
        runtimeSessionId=SESSION_ID,
        contentType="application/json",
        accept="application/vnd.amazon.eventstream",
        body={"command": command, "timeout": timeout},
    )
    exit_code = None
    for event in response.get("stream", []):
        chunk = event.get("chunk", {})
        if "contentDelta" in chunk:
            delta = chunk["contentDelta"]
            if delta.get("stdout"):
                print(delta["stdout"], end="")
            if delta.get("stderr"):
                print(delta["stderr"], end="", file=sys.stderr)
        if "contentStop" in chunk:
            exit_code = chunk["contentStop"].get("exitCode")
    return exit_code

# 1. 세션 시작 — 에이전트 호출
response = agentcore_client.invoke_agent_runtime(
    agentRuntimeArn=RUNTIME_ARN,
    runtimeSessionId=SESSION_ID,
    payload=json.dumps({
        "prompt": "작업 준비 완료를 알려 주세요"
    }).encode(),
)
print(f"세션 시작: {SESSION_ID}")

# 2. Shell Command — git clone
clone_exit = run_command(
    "git clone codecommit::us-west-2://dining-reservation /workspace/repo"
)
print(f"git clone exit: {clone_exit}")

# 3. 에이전트에게 코드 수정 요청 — 같은 세션이므로 파일시스템을 공유합니다
response = agentcore_client.invoke_agent_runtime(
    agentRuntimeArn=RUNTIME_ARN,
    runtimeSessionId=SESSION_ID,
    payload=json.dumps({
        "prompt": "repo/reservation.py에 예약 인원 검증 함수를 추가해 주세요"
    }).encode(),
)
body = json.loads(response["response"].read())
print(body["result"][:200])

# 4. Shell Command — git push
push_exit = run_command(
    "cd /workspace/repo && git add -A && "
    "git commit -m 'feat: add capacity validation' && git push"
)
print(f"git push exit: {push_exit}")

# --- console_app.py ---
import json
import uuid

import boto3
import streamlit as st

REGION = "us-west-2"
RUNTIME_ARN = "<YOUR_RUNTIME_ARN>"
BUCKET = "<YOUR_BUCKET_NAME>"  # coding-service-files-<YOUR_ACCOUNT_ID>

agentcore_client = boto3.client("bedrock-agentcore", region_name=REGION)
s3_client = boto3.client("s3", region_name=REGION)

st.set_page_config(page_title="팀 코딩 콘솔", page_icon="🛠️")
st.title("🛠️ 팀 코딩 콘솔")

def load_work_log() -> list[dict]:
    """S3에서 작업 이력을 읽어 시간 역순으로 반환합니다."""
    try:
        obj = s3_client.get_object(Bucket=BUCKET, Key="work-log.json")
        logs = json.loads(obj["Body"].read())
    except s3_client.exceptions.NoSuchKey:
        return []
    return sorted(logs, key=lambda e: e["timestamp"], reverse=True)

request_text = st.text_area(
    "코딩 요청",
    placeholder="예: reservation.py에 영업시간 검증을 추가해 주세요",
)

if st.button("실행", type="primary") and request_text.strip():
    with st.spinner("에이전트가 작업 중입니다..."):
        response = agentcore_client.invoke_agent_runtime(
            agentRuntimeArn=RUNTIME_ARN,
            runtimeSessionId=f"console-{uuid.uuid4().hex}",
            payload=json.dumps({"prompt": request_text}).encode(),
        )
        body = json.loads(response["response"].read())
    st.success("완료")
    st.markdown(body["result"])

st.divider()
st.subheader("작업 이력")

logs = load_work_log()
if logs:
    st.dataframe(
        [
            {
                "시간": entry["timestamp"],
                "요청": entry["request"][:60],
                "결과 요약": entry["result"][:80],
            }
            for entry in logs
        ],
        use_container_width=True,
        hide_index=True,
    )
else:
    st.info("아직 작업 이력이 없습니다. 위에서 요청을 실행해 보세요.")

도전 과제 방향
GitHub 연동: Secrets Manager에 PAT를 저장하고 Runtime 환경 변수로 주입한 뒤, 셸 명령에서 gh auth login --with-token으로 인증하면 gh pr create까지 같은 세션에서 처리됩니다
멀티세션 협업: 서로 다른 runtimeSessionId로 동시 호출한 뒤 각 세션에서 /workspace에 다른 파일을 만들어 교차 조회하면 격리를 실증할 수 있습니다 — 공유가 필요한 데이터만 /mnt/persistent에 둡니다
Slack 알림: work-log append 직후 Slack 웹훅을 호출하되, 웹훅 URL은 Secrets Manager에 두고 실패해도 본 작업이 롤백되지 않도록 예외를 삼킵니다
핵심 포인트
AgentCore Runtime은 세션마다 격리된 MicroVM을 제공하므로 팀원이 동시에 요청해도 간섭이 없습니다
runtimeSessionId는 호출자가 만들어 전달하는 멱등 키이므로, 같은 값을 넘기면 대화와 셸 명령이 같은 세션·파일시스템을 공유합니다
InvokeAgentRuntimeCommand의 출력은 이벤트 스트림으로 오므로 contentDelta로 진행 상황을 보고 contentStop의 exitCode로 성공을 판정합니다
S3 Files 마운트는 세션 수명과 무관하게 데이터를 보존하므로 작업 이력 추적과 감사에 적합합니다
Runtime 배포 후에는 IAM 권한만으로 호출이 제어되므로 별도 인증 서버 없이 팀 서비스로 운영할 수 있습니다

Lambda MicroVMs 코드 샌드박스
고급
코딩 에이전트가 실행하는 코드를 격리된 Lambda MicroVM 안에서 돌려, 호스트 접근 불가·테넌트 간 격리·종료 시 상태 소멸을 실증합니다.

미션 브리핑
앞 미션에서 코딩 에이전트는 Runtime 안에서 코드를 실행했지만, 악의적·실수적 코드가 호스트에 영향을 줄 위험은 남아 있습니다. 이번 미션에서는 Lambda MicroVM을 코드 실행 샌드박스로 도입합니다. MicroVM은 Firecracker 전용 커널 위에서 돌아가므로 호스트 파일시스템 접근이 원천 차단되고, 사용자마다 전용 환경이 배정됩니다. 종료하면 메모리·디스크 상태가 함께 사라져 흔적이 남지 않습니다. Tester 에이전트가 MicroVM의 전용 HTTPS 엔드포인트로 pytest를 실행하고 결과를 받아오는 엔드투엔드 연동까지 구축합니다.

# --- sandbox_server.py ---
import subprocess
import sys
from pathlib import Path

from fastapi import FastAPI
from pydantic import BaseModel

SANDBOX = Path("/sandbox")
app = FastAPI()

class TestRequest(BaseModel):
    source_code: str
    source_filename: str
    test_code: str

@app.get("/health")
def health():
    """샌드박스 준비 상태와 런타임 버전을 반환합니다."""
    pytest_version = subprocess.run(
        [sys.executable, "-m", "pytest", "--version"],
        capture_output=True, text=True,
    ).stdout.strip()
    return {"status": "ok", "python": sys.version.split()[0], "pytest": pytest_version}

@app.post("/run-tests")
def run_tests(request: TestRequest):
    """소스와 테스트를 /sandbox에 쓰고 pytest를 실행합니다."""
    (SANDBOX / request.source_filename).write_text(request.source_code)
    (SANDBOX / "test_code.py").write_text(request.test_code)

    completed = subprocess.run(
        [sys.executable, "-m", "pytest", "test_code.py", "--tb=short", "-q"],
        cwd=SANDBOX, capture_output=True, text=True, timeout=300,
    )
    return {"output": completed.stdout + completed.stderr, "exitCode": completed.returncode}

# --- sandbox_runner.py ---
import time

import boto3
import requests

REGION = "us-west-2"
IMAGE_ARN = "<YOUR_IMAGE_ARN>"
# 커넥터는 이름이 아니라 ARN입니다
ALL_INGRESS_ARN = f"arn:aws:lambda:{REGION}:aws:network-connector:aws-network-connector:ALL_INGRESS"
SHELL_INGRESS_ARN = f"arn:aws:lambda:{REGION}:aws:network-connector:aws-network-connector:SHELL_INGRESS"

microvms_client = boto3.client("lambda-microvms", region_name=REGION)

def create_sandbox() -> tuple[str, str]:
    """MicroVM을 실행하고 (microvm_id, endpoint)를 반환합니다."""
    run_response = microvms_client.run_microvm(
        imageIdentifier=IMAGE_ARN,
        ingressNetworkConnectors=[ALL_INGRESS_ARN, SHELL_INGRESS_ARN],
        idlePolicy={
            "maxIdleDurationSeconds": 300,
            "suspendedDurationSeconds": 600,
            "autoResumeEnabled": True,
        },
    )
    microvm_id = run_response["microvmId"]
    endpoint = run_response["endpoint"]
    print(f"MicroVM started: {microvm_id} ({run_response['state']})")

    # PENDING → RUNNING 대기
    while microvms_client.get_microvm(microvmIdentifier=microvm_id)["state"] == "PENDING":
        time.sleep(2)
    return microvm_id, endpoint

def get_auth_token(microvm_id: str, port: int = 8080) -> str:
    """엔드포인트 호출용 인증 토큰을 발급합니다 (최대 60분)."""
    token_response = microvms_client.create_microvm_auth_token(
        microvmIdentifier=microvm_id,
        expirationInMinutes=30,
        allowedPorts=[{"port": port}],
    )
    return token_response["authToken"]["X-aws-proxy-auth"]

def run_tests_in_microvm(
    endpoint: str,
    auth_token: str,
    source_code: str,
    source_filename: str,
    test_code: str,
) -> dict:
    """MicroVM의 /run-tests를 호출해 pytest 결과를 받습니다."""
    response = requests.post(
        f"https://{endpoint}/run-tests",
        headers={"X-aws-proxy-auth": auth_token},
        json={
            "source_code": source_code,
            "source_filename": source_filename,
            "test_code": test_code,
        },
        timeout=310,
    )
    response.raise_for_status()
    return response.json()

def destroy_sandbox(microvm_id: str):
    """MicroVM을 종료합니다 — 메모리·디스크 상태가 함께 사라집니다."""
    microvms_client.terminate_microvm(microvmIdentifier=microvm_id)
    print(f"MicroVM terminated: {microvm_id}")

if __name__ == "__main__":
    microvm_id, endpoint = create_sandbox()
    auth_token = get_auth_token(microvm_id)

    health = requests.get(
        f"https://{endpoint}/health",
        headers={"X-aws-proxy-auth": auth_token},
        timeout=30,
    )
    print(f"/health {health.status_code}: {health.json()}")

    result = run_tests_in_microvm(
        endpoint=endpoint,
        auth_token=auth_token,
        source_code="",
        source_filename="sample.py",
        test_code="def test_add(): assert 1+1 == 2\n",
    )
    print(f"pytest output:\n{result['output']}")

    destroy_sandbox(microvm_id)

# --- tester_microvm.py ---
import re

from strands import Agent, tool
from strands.models import BedrockModel

from sandbox_runner import (
    create_sandbox,
    destroy_sandbox,
    get_auth_token,
    run_tests_in_microvm,
)

@tool
def run_tests_in_sandbox(
    test_code: str,
    source_code: str,
    source_filename: str,
) -> str:
    """MicroVM 샌드박스 안에서 pytest를 실행합니다."""
    microvm_id, endpoint = create_sandbox()
    try:
        auth_token = get_auth_token(microvm_id)
        result = run_tests_in_microvm(
            endpoint=endpoint,
            auth_token=auth_token,
            source_code=source_code,
            source_filename=source_filename,
            test_code=test_code,
        )
        return parse_pytest_output(result["output"])
    finally:
        destroy_sandbox(microvm_id)

def parse_pytest_output(output: str) -> str:
    """pytest 출력에서 passed/failed 수를 추출합니다."""
    passed_match = re.search(r"(\d+) passed", output)
    failed_match = re.search(r"(\d+) failed", output)

    passed = int(passed_match.group(1)) if passed_match else 0
    failed = int(failed_match.group(1)) if failed_match else 0

    summary = f"passed={passed} failed={failed}"

    if failed > 0:
        # 실패 로그 포함
        return f"{summary}\n\n--- 실패 상세 ---\n{output}"
    return summary

TESTER_SYSTEM_PROMPT = """당신은 Tester 에이전트입니다.
주어진 소스 코드에 대해 테스트를 작성하고 MicroVM 샌드박스에서 실행합니다.
경계값, 오류 케이스, 정상 케이스를 모두 커버하세요.
결과는 마지막 줄에 passed=N failed=M 형식으로 보고합니다."""

tester_model = BedrockModel(
    model_id="us.anthropic.claude-sonnet-4-6",
    region_name="us-west-2",
)

tester_agent = Agent(
    model=tester_model,
    system_prompt=TESTER_SYSTEM_PROMPT,
    tools=[run_tests_in_sandbox],
)

격리 축은 호스트 파일시스템·테넌트 간·종료 시 소멸 3가지입니다. 네트워크 아웃바운드는 기본적으로 퍼블릭 인터넷이 열려 있으므로, 차단을 확인하는 축이 아니라 VPC egress 커넥터로 통제하는 대상으로 다룹니다.

# --- test_isolation.py ---
import socket

import boto3
import requests

from sandbox_runner import create_sandbox, destroy_sandbox, get_auth_token

REGION = "us-west-2"
microvms_client = boto3.client("lambda-microvms", region_name=REGION)

def shell_exec(endpoint: str, auth_token: str, command: str) -> str:
    """샌드박스 서버의 /run-tests를 통해 셸 명령의 출력을 얻습니다."""
    response = requests.post(
        f"https://{endpoint}/run-tests",
        headers={"X-aws-proxy-auth": auth_token},
        json={
            "source_code": "",
            "source_filename": "probe.py",
            "test_code": (
                "import subprocess\n"
                "def test_probe():\n"
                f"    out = subprocess.run({command!r}, shell=True, capture_output=True, text=True)\n"
                "    print(out.stdout + out.stderr)\n"
                "    assert True\n"
            ),
        },
        timeout=60,
    )
    return response.json()["output"]

def test_filesystem_isolation():
    """호스트 파일시스템 격리 테스트."""
    print("\n=== 1. 호스트 파일시스템 격리 ===")

    microvm_id, endpoint = create_sandbox()
    auth_token = get_auth_token(microvm_id)

    host_hostname = socket.gethostname()
    microvm_hostname = shell_exec(endpoint, auth_token, "cat /etc/hostname")
    print(f"호스트 hostname: {host_hostname}")
    print(f"MicroVM hostname: {microvm_hostname.strip()}")
    assert host_hostname not in microvm_hostname, "hostname이 같으면 격리 실패"

    root_listing = shell_exec(endpoint, auth_token, "ls /")
    print(f"ls /:\n{root_listing}")

    mount_result = shell_exec(endpoint, auth_token, "mount | grep -v proc | grep -v sys")
    print(f"마운트 목록:\n{mount_result}")

    destroy_sandbox(microvm_id)
    print("✅ 파일시스템 격리 확인 완료")

def test_tenant_isolation():
    """MicroVM 간 격리 테스트."""
    print("\n=== 2. MicroVM 간 격리 ===")

    microvm_id_1, endpoint_1 = create_sandbox()
    microvm_id_2, endpoint_2 = create_sandbox()
    token_1 = get_auth_token(microvm_id_1)
    token_2 = get_auth_token(microvm_id_2)

    shell_exec(endpoint_1, token_1, "echo 'tenant-a' > /sandbox/evidence.txt")
    own = shell_exec(endpoint_1, token_1, "cat /sandbox/evidence.txt")
    print(f"1번 MicroVM - evidence.txt: {own.strip()[:80]}")
    assert "tenant-a" in own

    other = shell_exec(endpoint_2, token_2, "cat /sandbox/evidence.txt 2>&1")
    print(f"2번 MicroVM - evidence.txt: {other.strip()[:80]}")
    assert "tenant-a" not in other

    destroy_sandbox(microvm_id_1)
    destroy_sandbox(microvm_id_2)
    print("✅ 테넌트 간 격리 확인 완료")

def test_state_destruction():
    """종료 시 상태 소멸 테스트."""
    print("\n=== 3. 종료 시 상태 소멸 ===")

    microvm_id_1, endpoint_1 = create_sandbox()
    token_1 = get_auth_token(microvm_id_1)
    shell_exec(endpoint_1, token_1, "echo 'secret-data-12345' > /sandbox/evidence.txt")
    destroy_sandbox(microvm_id_1)
    print("1번 MicroVM 종료됨")

    microvm_id_2, endpoint_2 = create_sandbox()
    token_2 = get_auth_token(microvm_id_2)
    verify = shell_exec(endpoint_2, token_2, "cat /sandbox/evidence.txt 2>&1")
    print(f"새 MicroVM - evidence.txt: {verify.strip()[:80]}")
    assert "secret-data-12345" not in verify

    destroy_sandbox(microvm_id_2)
    print("✅ 상태 소멸 확인 완료 - 이전 상태 완전 소멸")

if __name__ == "__main__":
    test_filesystem_isolation()
    test_tenant_isolation()
    test_state_destruction()
    print("\n🎉 모든 격리 테스트 통과!")

# --- sandbox_console.py ---
import streamlit as st

from sandbox_runner import (
    create_sandbox,
    destroy_sandbox,
    get_auth_token,
    run_tests_in_microvm,
)
from test_isolation import shell_exec

st.set_page_config(page_title="MicroVM 샌드박스 콘솔", layout="wide")
st.title("MicroVM 샌드박스 콘솔")

col_source, col_test = st.columns(2)
source_code = col_source.text_area(
    "소스 코드", height=240, value="def validate_party_size(n):\n    return 1 <= n <= 20\n"
)
test_code = col_test.text_area(
    "테스트 코드",
    height=240,
    value="from reservation import validate_party_size\n\n"
          "def test_ok(): assert validate_party_size(4)\n"
          "def test_too_many(): assert not validate_party_size(21)\n",
)

if st.button("MicroVM에서 실행", type="primary"):
    with st.spinner("MicroVM 실행 중..."):
        microvm_id, endpoint = create_sandbox()
        auth_token = get_auth_token(microvm_id)
        try:
            result = run_tests_in_microvm(
                endpoint=endpoint,
                auth_token=auth_token,
                source_code=source_code,
                source_filename="reservation.py",
                test_code=test_code,
            )
            hostname = shell_exec(endpoint, auth_token, "cat /etc/hostname").strip()
        finally:
            destroy_sandbox(microvm_id)

    st.subheader("pytest 출력")
    st.code(result["output"], language="text")
    st.caption(f"exit code: {result['exitCode']}")

    # 격리 증거 — 실행마다 microvmId와 hostname이 달라집니다
    st.sidebar.subheader("격리 증거")
    st.sidebar.write(f"**microvmId**: `{microvm_id}`")
    st.sidebar.write("**상태**: TERMINATED (실행 후 즉시 종료)")
    st.sidebar.write(f"**/etc/hostname**: `{hostname}`")

도전 과제 방향
라이프사이클 훅: hooks의 microvmImageHooks에서 ready와 validate를 ENABLED로 켜고, 샌드박스 서버에 두 경로를 추가합니다. /validate에서 모의 pytest를 한 번 돌리면 Lambda가 그 구간의 스냅샷을 프리페치해 시작이 빨라집니다
VPC egress 통제: create_network_connector로 VpcEgressConfiguration(서브넷·보안 그룹·AssociatedComputeResourceTypes: ["MicroVm"]) 커넥터를 만들고, run_microvm의 egressNetworkConnectors에 ARN을 넣습니다. 보안 그룹 아웃바운드 규칙이 그대로 적용됩니다
suspend/resume 활용: 테스트마다 종료하는 대신 suspend_microvm으로 멈추고 다음 요청에 resume_microvm합니다. pip install 캐시가 유지되는지 비교하면 스냅샷 재개의 이득이 보입니다 - 단, 커넥터는 실행 시점에 고정되어 재개 시 바꿀 수 없습니다
핵심 포인트
MicroVM은 Firecracker 전용 커널에서 실행되므로 호스트와 완전히 격리됩니다 - 컨테이너보다 강력한 보안 경계를 제공합니다
이미지는 Dockerfile을 API에 붙여 넣는 방식이 아니라 zip 아티팩트를 S3로 넘기는 방식입니다 - 빌드는 Lambda가 수행하고 결과는 스냅샷으로 남습니다
접속은 전용 HTTPS 엔드포인트 + X-aws-proxy-auth 토큰 조합입니다 - 토큰이 MicroVM·포트·만료 시각에 묶여 있어 최소 권한 접근이 됩니다
종료하면 메모리와 디스크 상태가 함께 사라지므로 이전 실행의 민감 데이터가 다음 실행에 노출될 위험이 없습니다
아웃바운드는 기본적으로 퍼블릭 인터넷이 열려 있습니다 - 통제가 필요하면 VPC egress 네트워크 커넥터를 붙이고 보안 그룹으로 좁힙니다
