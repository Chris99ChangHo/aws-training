시리즈 5: AgentCore 통합
시리즈 개요
「AgentCore 배포」에서 다이닝 에이전트를 Runtime에 올렸습니다. 배포된 에이전트는 아직 몸만 있는 상태입니다 — 이 시리즈에서 능력을 붙입니다. Gateway에 KB 검색 Lambda 타깃을 연결해 식당 지식을 도구로 제공하고, AgentCore Memory로 "이탈리안을 좋아하고 매운 음식을 못 먹어요" 같은 취향을 세션 간 기억시키고, 마지막으로 Runtime·Gateway·Memory를 모두 쓰는 Streamlit 프로덕션 다이닝 앱으로 통합합니다.

# lambda_function.py
import os
import boto3

agent_rt = boto3.client("bedrock-agent-runtime", region_name="us-west-2")

def lambda_handler(event, context):
    tool_name = context.client_context.custom["bedrockAgentCoreToolName"].split("___")[-1]
    if tool_name == "search_knowledge_base":
        resp = agent_rt.retrieve(
            knowledgeBaseId=os.environ["KB_ID"],
            retrievalQuery={"text": event["query"]},
        )
        chunks = [r["content"]["text"] for r in resp["retrievalResults"]]
        return {"content": [{"type": "text", "text": "\n\n".join(chunks)}]}
    return {"content": [{"type": "text", "text": f"알 수 없는 도구: {tool_name}"}]}

# app/DiningConcierge/main.py — 엔트리포인트 안에서 결선
from strands import Agent
from strands.models import BedrockModel
from strands.tools.mcp import MCPClient
from mcp_proxy_for_aws.client import aws_iam_streamablehttp_client
from bedrock_agentcore import BedrockAgentCoreApp

# 💡 선행 미션의 식당 도구 — check_reservations는 로컬 MCP 서버 의존이라 제외
from tools import search_restaurants, get_menu

GATEWAY_URL = "https://<YOUR_GATEWAY_ID>.gateway.bedrock-agentcore.us-west-2.amazonaws.com/mcp"

model = BedrockModel(model_id="us.anthropic.claude-sonnet-4-6")
app = BedrockAgentCoreApp()

@app.entrypoint
def invoke(payload):
    gateway_client = MCPClient(
        lambda: aws_iam_streamablehttp_client(
            endpoint=GATEWAY_URL,
            aws_region="us-west-2",
            aws_service="bedrock-agentcore",
        )
    )
    with gateway_client:
        # 💡 Gateway 도구 + 선행 미션의 로컬 식당 도구를 함께 사용합니다
        agent = Agent(
            model=model,
            tools=gateway_client.list_tools_sync() + [search_restaurants, get_menu],
        )
        return {"result": str(agent(payload.get("prompt", "")))}

if __name__ == "__main__":
    app.run()

# app.py
import json
import uuid
import boto3
import streamlit as st

client = boto3.client("bedrock-agentcore", region_name="us-west-2")
RUNTIME_ARN = "<YOUR_RUNTIME_ARN>"

st.title("🍽️ 다이닝 컨시어지")

if "session_id" not in st.session_state:
    st.session_state.session_id = str(uuid.uuid4())
if "messages" not in st.session_state:
    st.session_state.messages = []

for msg in st.session_state.messages:
    st.chat_message(msg["role"]).write(msg["content"])

if prompt := st.chat_input("질문을 입력하세요"):
    st.session_state.messages.append({"role": "user", "content": prompt})
    st.chat_message("user").write(prompt)

    response = client.invoke_agent_runtime(
        agentRuntimeArn=RUNTIME_ARN,
        runtimeSessionId=st.session_state.session_id,
        payload=json.dumps({"prompt": prompt}).encode("utf-8"),
    )
    # 💡 응답은 response 블롭 — read()로 꺼내 JSON 파싱합니다
    body = response["response"].read().decode("utf-8")
    try:
        result_text = json.loads(body).get("result", body)
    except json.JSONDecodeError:
        result_text = body

    st.session_state.messages.append({"role": "assistant", "content": result_text})
    st.chat_message("assistant").write(result_text)

Web Search 타깃 추가: Gateway 콘솔에서 target protocol을 MCP target, target type을 Connectors로 지정하면 사전 구성된 Web Search tool을 타깃으로 붙일 수 있습니다 — 파라미터(domainFilter 등)를 코드로 다루는 절차는 AgentCore Web Search 실습에서 다룹니다.
두 번째 Lambda 타깃: 같은 Gateway에 타깃을 하나 더 추가하면 에이전트 코드는 그대로인 채 list_tools의 도구 수만 늘어납니다 — 새 타깃도 스키마(name/description/inputSchema)를 명시해야 MCP 도구 목록에 나타납니다.
자동화: 콘솔로 만든 구성을 boto3(create_gateway·create_gateway_target)로 재현하는 코드는 AgentCore Gateway 실습의 심화 과제에 있습니다 — Gateway 역할은 콘솔의 Create default role이 만들어 주던 것을 직접 준비해야 합니다.

# dining_agent_memory.py — Memory 연동 강남 식당 추천 에이전트
from strands import Agent, tool
from strands.models import BedrockModel
from bedrock_agentcore.memory.integrations.strands.config import (
    AgentCoreMemoryConfig,
    RetrievalConfig,
)
from bedrock_agentcore.memory.integrations.strands.session_manager import (
    AgentCoreMemorySessionManager,
)

REGION = "us-west-2"
MEMORY_ID = "<YOUR_MEMORY_ID>"                        # 단계 3에서 복사한 값
SEMANTIC_STRATEGY_ID = "<YOUR_SEMANTIC_STRATEGY_ID>"  # 단계 3에서 확인한 SEMANTIC 전략 ID

# --- 도구 ---
@tool
def search_restaurants(location: str, cuisine: str) -> str:
    """식당을 검색합니다."""
    data = {
        "이탈리안": "트라토리아 벨라 (강남역 3분, 45,000원, ★4.5)",
        "한식": "한우명가 (역삼역 5분, 65,000원, ★4.3)",
        "프렌치": "르 비스트로 (압구정역 2분, 90,000원, ★4.6)",
    }
    return data.get(cuisine, f"{cuisine} 식당을 찾지 못했습니다.")

# --- 에이전트 생성 함수 — session_id/actor_id를 바꿔 끼울 수 있게 ---
def create_agent(session_id: str, actor_id: str = "user-001") -> Agent:
    #      키 = 검색할 네임스페이스(빌트인 기본형: /strategy/{전략ID}/actors/{actorId}/)
💡
    #      {actorId}는 실행 시 actor_id 값으로 자동 치환됩니다
    config = AgentCoreMemoryConfig(
        memory_id=MEMORY_ID,
        actor_id=actor_id,
        session_id=session_id,
        retrieval_config={
            "/strategy/" + SEMANTIC_STRATEGY_ID + "/actors/{actorId}/":
                RetrievalConfig(top_k=5, relevance_score=0.3),
        },
    )
    session_manager = AgentCoreMemorySessionManager(
        agentcore_memory_config=config,
        region_name=REGION,
    )
    return Agent(
        model=BedrockModel(model_id="us.anthropic.claude-sonnet-4-6"),
        system_prompt="강남 식당 추천 전문가. 사용자의 이전 취향을 기억해 반영합니다.",
        tools=[search_restaurants],
        session_manager=session_manager,  # ← Memory 연결
        callback_handler=None,  # 스트리밍 중간 출력 억제 — 최종 print만 표시
    )

if __name__ == "__main__":
    agent = create_agent(session_id="session-dining-000")
💡
    print(agent("강남에서 어떤 식당을 추천해 줄 수 있나요?"))

도전 과제 방향
지난번 그 식당: 세션 1의 대화에 추천받은 식당 이름이 남아 있어야 semantic 전략이 사실로 추출합니다 — 같은 actor_id로 새 session_id를 발급해(create_agent) 회상 질문을 보냅니다. 추출이 비동기라 1~2분 대기가 필요하고, 회상이 걸러지면 RetrievalConfig의 relevance_score(0.3)를 낮춰 확인하세요.
멀티 사용자: create_agent의 actor_id를 user-002 등으로 바꾸면 됩니다 — 장기 기억 네임스페이스의 {actorId} 치환이 고객별 격리를 만듭니다. 한 사용자의 취향 입력이 다른 사용자의 추천에 섞이지 않는지 교차로 확인하세요. 같은 방향이 AgentCore Memory 연동 실습의 도전 과제에 있습니다.
Streamlit 연결: 단계 6의 app.py가 골격입니다 — 사이드바에 사용자 선택 UI를 더해 선택값을 create_agent의 actor_id로 전달하면 고객별 개인화 다이닝 앱이 됩니다. 상세 구성은 후속 미션이 다룹니다.

핵심 포인트
AgentCore Memory는 대화 원본을 단기 기억(Event)으로 저장하고, semantic 전략이 백그라운드에서 핵심 사실을 장기 기억으로 추출합니다 — eventExpiryDuration(이벤트 보존 기간)은 단기 기억에만 적용됩니다
AgentCoreMemorySessionManager는 Strands Agent의 session_manager에 연결하여 대화를 자동으로 Memory에 저장합니다 — 장기 기억 검색은 retrieval_config를 지정해야 동작합니다(미지정 시 저장만 됨)
빌트인 전략의 장기 기억은 /strategy/{전략ID}/actors/{actorId}/ 네임스페이스에 저장됩니다 — 그래서 전략 ID 확인 단계가 필요합니다
같은 actor_id + 다른 session_id → 이전 세션에서 추출된 취향이 자동으로 컨텍스트에 주입됩니다
장기 기억 추출은 비동기로 처리되므로 세션 간 테스트 시 1~2분 정도 대기가 필요합니다

# agent_client.py — 배포된 DiningConcierge Runtime 호출 클라이언트
import json

import boto3

REGION = "us-west-2"
RUNTIME_ARN = "<YOUR_AGENT_RUNTIME_ARN>"  # 단계 2에서 복사한 값

client = boto3.client("bedrock-agentcore", region_name=REGION)
control = boto3.client("bedrock-agentcore-control", region_name=REGION)

def invoke_agent(prompt: str, session_id: str, actor_id: str) -> dict:
    # 💡: runtimeSessionId는 33자 이상 — 같은 값은 같은 세션(멱등)
    response = client.invoke_agent_runtime(
        agentRuntimeArn=RUNTIME_ARN,
        runtimeSessionId=session_id,
        qualifier="DEFAULT",
        payload=json.dumps(
            {"prompt": prompt, "session_id": session_id, "actor_id": actor_id}
        ).encode("utf-8"),
    )
    body = response["response"].read().decode("utf-8")
    try:
        return json.loads(body)  # {"result": ..., "tool_calls": [...]}
    except json.JSONDecodeError:
        return {"result": body, "tool_calls": []}

def get_runtime_status() -> str:
    for rt in control.list_agent_runtimes()["agentRuntimes"]:
        if rt["agentRuntimeName"] == "DiningConcierge":
            return rt["status"]
    return "NOT_FOUND"

if __name__ == "__main__":
    print("Runtime:", get_runtime_status())
    print(invoke_agent("강남 식당 추천해 주세요", "session-app-test-000000000000000000", "user-001"))

# memory_client.py — Memory 취향 조회 클라이언트
from bedrock_agentcore.memory import MemoryClient

REGION = "us-west-2"
MEMORY_ID = "<YOUR_MEMORY_ID>"                        # 단계 2에서 복사한 값
SEMANTIC_STRATEGY_ID = "<YOUR_SEMANTIC_STRATEGY_ID>"  # 단계 2에서 복사한 값

client = MemoryClient(region_name=REGION)

def get_preferences(actor_id: str) -> list[str]:
    # 💡: 빌트인 전략의 장기 기억 네임스페이스 — /strategy/{전략ID}/actors/{actorId}/
    memories = client.retrieve_memories(
        memory_id=MEMORY_ID,
        namespace=f"/strategy/{SEMANTIC_STRATEGY_ID}/actors/{actor_id}/",
        query="고객의 음식 취향",
    )
    return [m["content"]["text"] for m in memories]

if __name__ == "__main__":
    print(get_preferences("user-001"))

도전 과제 방향
Web Search 근거 반영: Gateway에 Connectors > Web Search 타깃을 추가하면 실시간 정보(날씨·이벤트) 근거를 추천에 반영하고 도구 로그에 web-search 호출을 남길 수 있습니다. 단, Web Search 커넥터는 us-east-1 전용이라 이 시리즈의 Gateway(us-west-2)에는 추가할 수 없습니다 — us-east-1 리전에 Gateway를 만들면 시도할 수 있습니다. 절차는 AgentCore Web Search 실습 참고.
스트리밍 응답: 엔트리포인트를 async 스트리밍으로 바꾸면 응답 본문에 SSE 프레이밍(data: 줄)이 섞입니다 — 줄 단위로 파싱해 도착하는 대로 표시합니다.
Observability 연동: CloudWatch에서 Runtime 호출 트레이스를 확인해 도구 호출 구간별 소요를 살펴봅니다.

핵심 포인트
프로덕션 앱은 CLI가 아니라 boto3 bedrock-agentcore의 invoke_agent_runtime으로 Runtime을 호출합니다 — 응답 블롭 하나에 result와 tool_calls를 함께 담아 앱이 파싱합니다
취향 저장은 Runtime 쪽(session_manager)이, 취향 표시는 앱 쪽(MemoryClient.retrieve_memories)이 담당합니다 — 같은 MEMORY_ID·네임스페이스를 바라봐야 연결됩니다
session_id는 대화 단위, actor_id는 사용자 단위 — "새 대화 시작"은 session_id만 갈아 끼우므로 장기 기억(취향)이 유지됩니다
Runtime(에이전트) + Gateway(도구) + Memory(기억)를 각자 교체·확장할 수 있는 조립 구조가 이 시리즈의 최종 아키텍처입니다 — 앱 코드는 호출과 표시만 담당합니다

관련 실습
AgentCore CLI 배포 — Strands 에이전트를 AgentCore Runtime에 배포하고, 호출 검증과 미니 챗 UI까지 확인합니다
AgentCore Gateway 연결 — Gateway에 Lambda Target을 등록하고 MCP로 호출합니다
AgentCore Memory 연동 — Memory를 연동하여 에이전트가 대화를 기억하도록 합니다
AgentCore Web Search — Gateway Web Search 커넥터로 실시간 웹 검색
