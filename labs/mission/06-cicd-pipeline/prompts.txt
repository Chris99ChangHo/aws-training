시리즈 6: CI/CD
시리즈 개요
손으로 배포하던 다이닝 에이전트를 파이프라인에 태웁니다. 전반부는 배포 전 안전장치입니다 — 소스가 바뀔 때마다 Strands Evals가 응답 품질을 채점하고, 임계를 넘어야만 AgentCore에 자동 배포되는 게이트를 만들고, 나쁜 버전으로 그 게이트가 정말 막는지 실증합니다. 후반부는 배포 후 안전장치입니다 — 배포된 새 버전을 canary 엔드포인트에서 검증하고 통과할 때만 production을 자동 승격하는 카나리 스테이지를 붙인 뒤, 게이트는 통과하지만 실환경에서만 드러나는 결함을 카나리가 잡아내는 것까지 확인합니다.

미션 브리핑
지금까지 다이닝 에이전트는 손으로 배포해 왔습니다. 이제 코드가 바뀔 때마다 사람 손을 거치지 않고 — 단, 평가 게이트를 통과해야만 — 자동 배포되게 만듭니다. 소스 버킷에 새 버전을 올리면 파이프라인이 Strands Evals로 응답 품질을 채점하고, 평균 점수가 임계를 넘어야 AgentCore에 배포합니다. 이 미션은 파이프라인을 조립해 첫 실행을 성공시킨 뒤, 파이프라인이 배포한 에이전트를 미니 Streamlit 앱에서 호출해 소스 변경이 앱 응답까지 반영되는 것을 눈으로 확인하는 것까지입니다 — 게이트가 실제로 배포를 막는 실증은 다음 미션에서 이어집니다.

# --- DiningConcierge/app/DiningConcierge/main.py ---
from bedrock_agentcore import BedrockAgentCoreApp
from strands import Agent, tool
from strands.models import BedrockModel

SYSTEM_PROMPT = """당신은 강남 다이닝 컨시어지 AI 어시스턴트입니다.
강남 일대 식당 추천, 메뉴 안내, 예약 문의를 도와줍니다.
모르는 정보는 추측하지 말고 '확인 후 안내하겠습니다'라고 답하세요."""

CATALOG = [
    {"name": "트라토리아 벨라", "cuisine": "이탈리안", "location": "강남역", "spicy": "보통"},
    {"name": "스시 오마카세 하루", "cuisine": "일식", "location": "강남역", "spicy": "순함"},
    {"name": "한우명가", "cuisine": "한식", "location": "역삼역", "spicy": "보통"},
    {"name": "매콤한 마라", "cuisine": "중식", "location": "강남역", "spicy": "아주 매움"},
]

@tool
def search_restaurants(query: str) -> str:
    """강남 식당 카탈로그를 검색합니다."""
    results = [r for r in CATALOG if query.lower() in str(r).lower()]
    if not results:
        results = CATALOG
    return str(results)

model = BedrockModel(model_id="us.anthropic.claude-sonnet-4-6")
agent = Agent(model=model, system_prompt=SYSTEM_PROMPT, tools=[search_restaurants])

app = BedrockAgentCoreApp()

@app.entrypoint
def my_agent(payload):
    result = agent(payload.get("prompt", ""))
    return {"result": str(result)}

"""evals/gate_eval.py — 평가 케이스 3종 평균 0.7 미만이면 exit 1로 차단"""
import sys
from pathlib import Path

# zip 루트 기준 DiningConcierge 프로젝트를 import 경로에 추가
sys.path.insert(0, str(Path(__file__).resolve().parent.parent / "DiningConcierge"))

from strands import Agent
from strands.models import BedrockModel
from strands_evals import Case, Experiment
from strands_evals.evaluators import OutputEvaluator

from app.DiningConcierge.main import SYSTEM_PROMPT, search_restaurants

THRESHOLD = 0.7

def run_agent(case: Case) -> str:
    """배포될 코드와 같은 프롬프트·도구 구성으로 에이전트를 호출합니다."""
    agent = Agent(
        model=BedrockModel(model_id="us.anthropic.claude-sonnet-4-6", region_name="us-west-2"),
        system_prompt=SYSTEM_PROMPT,
        tools=[search_restaurants],
        callback_handler=None,
    )
    return str(agent(case.input))

cases = [
    Case[str, str](
        name="italian-recommend",
        input="강남역 근처 이탈리안 식당 추천해 주세요",
        expected_output="트라토리아 벨라를 추천하고 근거를 함께 안내합니다.",
    ),
    Case[str, str](
        name="no-spicy",
        input="매운 음식을 못 먹는데 강남 식당 추천해 주세요",
        expected_output="맵지 않은 식당만 추천하고, 매운맛 전문점인 매콤한 마라는 추천에 포함하지 않습니다.",
    ),
    Case[str, str](
        name="no-guessing",
        input="다음 주 공휴일에 영업하나요?",
        expected_output="확인되지 않은 정보를 추측하지 않고, 지금은 확인이 불가하다고 안내합니다.",
    ),
]

rubric = """
Evaluate whether the actual output satisfies the expected behavior:
1. REQUIRED FACTS: the restaurant and reasons in the expected output are present
2. EXCLUSIONS: items the expected output says to exclude are NOT recommended
3. NO GUESSING: unverifiable questions are not answered with guesses
Score 1.0 if fully satisfied, 0.5 if partial, 0.0 if violated.
"""

experiment = Experiment[str, str](
    cases=cases,
    evaluators=[OutputEvaluator(rubric=rubric, include_inputs=True)],
)
report = experiment.run_evaluations(run_agent)

avg = sum(report.scores) / len(report.scores)
print(f"평균 점수: {avg:.2f} (임계 {THRESHOLD})")
sys.exit(0 if avg >= THRESHOLD else 1)

# --- app.py (미니 Streamlit 앱) ---
import streamlit as st
import boto3
import json

REGION = "us-west-2"
RUNTIME_ARN = "<YOUR_RUNTIME_ARN>"

client = boto3.client("bedrock-agentcore", region_name=REGION)

st.title("🍽️ 다이닝 컨시어지")
user_input = st.chat_input("질문을 입력하세요")

if user_input:
    st.chat_message("user").write(user_input)
    response = client.invoke_agent_runtime(
        agentRuntimeArn=RUNTIME_ARN,
        payload=json.dumps({"prompt": user_input}).encode(),
    )
    body = json.loads(response["response"].read())
    st.chat_message("assistant").write(body.get("result", ""))

핵심 포인트
게이트의 본질은 평가를 종료 코드로 바꾸는 것 — CodeBuild는 비0 종료 코드를 스테이지 실패로 취급
S3 버전드 소스는 같은 키 재업로드만으로 새 실행을 트리거
임계 0.7은 품질 목표가 아니라 차단선 — 케이스 3종 중 1종 실패(평균 약 0.67)만으로 배포 차단

핵심 포인트
결함 만들기는 규칙 삭제가 아니라 역지시 교체 — 삭제만으로는 모델 기본 성향으로 통과 가능
로그 판독까지가 한 세트 — 실패 케이스와 점수가 다음 수정 명세가 됨
게이트 차단 시 Runtime은 이전 버전 그대로 — 앱 응답 불변이 사용자 관점의 증거
